\begin{document}
\begin{tikzpicture}[
    >=Stealth, node distance=10mm,

  \node[box, below=of stop] (idx) {BM25S index build\\$k_1{=}0.9,\ b{=}0.4$};
  \node[data, below=of idx] (index) {Inverted index\\(\textasciitilde 5\,GB)};

  \draw[arrow] (corpus) -- (tok);
  \draw[arrow] (tok) -- (stop);
  \draw[arrow] (stop) -- (idx);
